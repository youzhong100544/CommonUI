在 JavaScript 中，当一个 Promise 同时有 .then() 和 .catch()，并且该 Promise 被 reject（拒绝） 时，会走 .catch() 方法。
⚠️ 重要：如果 .then() 的第二个参数处理了错误，.catch() 就不会执行





应用

使用 Promise 遍历数组并动态添加任务执行

在 JavaScript 中，使用 Promise 遍历数组并动态添加任务执行 是一个常见需求，比如：

  • 依次执行多个异步操作（串行）
  • 并行执行多个异步操作（并发）
  • 动态根据条件添加新任务

🚫 常见错误


			// ❌ 错误：forEach 不能 await
			urls.forEach(async url => {
				await loadScript(url); // 实际是并行执行！
			});
		

			// ❌ 错误：map + await 不等于串行
			const promises = urls.map(url => loadScript(url));
			await promises; // 还是并行

正确串行必须用 for...of 或 reduce



✅ 场景 1：并行执行（所有任务同时开始）

使用 Promise.all()：


			const urls = ['a.js', 'b.js', 'c.js'];

			function loadScript(src) {
				return new Promise((resolve, reject) => {
					const script = document.createElement('script');
					script.src = src;
					script.onload = () => resolve(src);
					script.onerror = () => reject(new Error(`Failed to load ${src}`));
					document.head.appendChild(script);
				});
			}

			// 并行加载所有脚本
			Promise.all(urls.map(url => loadScript(url)))
			.then(results => {
				console.log('所有脚本加载完成:', results); // ['a.js', 'b.js', 'c.js']
				// 继续执行后续逻辑
			})
			.catch(err => {
				console.error('某个脚本加载失败:', err);
			});

✅ 优点：快
❌ 缺点：如果一个失败，整个 Promise.all 会 reject（除非用 Promise.allSettled）



✅ 场景 2：串行执行（一个接一个）

使用 async/await + for...of：


			async function loadScriptsInOrder(urls) {
				const results = [];
				for (const url of urls) {
					try {
						const result = await loadScript(url);
						results.push(result);
						console.log(`${url} 加载完成`);
					} catch (err) {
						console.error(`${url} 加载失败:`, err);
						// 可选择继续 or 中断
						// throw err; // 中断
					}
				}
				return results;
			}

			// 使用
			loadScriptsInOrder(['a.js', 'b.js', 'c.js'])
			.then(() => {
				console.log('所有脚本按顺序加载完成');
			});

✅ 适合有依赖关系的任务（比如 b.js 依赖 a.js）



✅ 场景 3：动态添加任务（在执行过程中追加）

这是你问题中的关键：“动态添加任务”

方法：使用一个“任务队列” + async 函数


			let taskQueue = ['a.js', 'b.js']; // 初始任务
			const executed = new Set();

			async function runTasks() {
				while (taskQueue.length > 0) {
					const current = taskQueue.shift(); // 取出第一个任务
					
					if (executed.has(current)) continue; // 避免重复执行
					executed.add(current);

					try {
						await loadScript(current);
						console.log(`${current} 执行完成`);

						// ✨ 动态添加新任务！
						if (current === 'b.js') {
							taskQueue.push('c.js', 'd.js'); // 动态追加
						}
					} catch (err) {
						console.error(`任务 ${current} 失败:`, err);
					}
				}
				console.log('所有任务（包括动态添加的）执行完毕');
			}

			// 启动
			runTasks();

✅ 这种方式支持在执行过程中动态扩展任务列表，非常灵活。



✅ 场景 5：带节流的并发执行（限制同时执行数量）

比如：最多同时加载 2 个脚本


			async function loadWithConcurrencyLimit(tasks, limit = 2) {
				const executing = [];

				for (const task of tasks) {
					const promise = loadScript(task).finally(() => {
						const index = executing.indexOf(promise);
						executing.splice(index, 1);
					});
					executing.push(promise);

					if (executing.length >= limit) {
						await Promise.race(executing); // 等待任意一个完成
					}
				}

				await Promise.all(executing); // 等待剩余任务完成
			}

			// 使用
			loadWithConcurrencyLimit(['a.js', 'b.js', 'c.js', 'd.js'], 2);
