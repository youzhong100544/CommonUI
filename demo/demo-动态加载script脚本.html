动态加载 script 脚本





基础知识

				let script = document.createElement('script');
				script.setAttribute('type', 'text/javascript');
				// script.setAttribute('src', './js/test.js');
				script.setAttribute('src', src);

			function loadScriptElement(src) {
				var script = document.createElement('script');
				script.type = 'text/javascript';
				// script.src = './js/test.js';
				script.src = src;
				// script.async = true;
				// script.defer = true;
				script.onload = function () {
					console.log('Script loaded successfully!');
				};
				script.onerror = () => {
					console.log(`Script loaded failure!`);
				};
				document.head.appendChild(script);
			}

应用

在 a.js 中动态添加 b.js 脚本文件，并等待它加载完成后再继续执行后续代码

🚫 错误做法（不要这样写）：


			// ./js/a.js
			// ❌ 不等待加载，b.js 可能还没执行
			const script = document.createElement('script');
			script.type = 'text/javascript';
			script.src = '/js/b.js';
			document.head.appendChild(script);

			someFunctionFromB(); // ⛔ 可能报错：not defined

方式一


			// ./js/a.js
			function loadScriptElement() {

				const script = document.createElement('script');

				script.type = 'text/javascript';
				// script.type = 'module';
				script.src = './js/b.js';
				// script.async = true; // 可选：异步加载，不阻塞页面
				// script.defer = true;

				script.onload = function() {
					console.log('b.js 加载完成');

					// 👇 在这里写依赖 b.js 的代码
					// someFunctionFromB();
				};

				script.onerror = function() {
					console.error('b.js 加载失败');
				};

				document.head.appendChild(script);
			}

方式二


			// ./js/a.js
			function loadScript(src) {
				return new Promise((resolve, reject) => {
					const script = document.createElement('script');
					
					script.type = 'text/javascript';
					// script.type = 'module';
					script.src = src;
					// script.src = './js/b.js';
					// script.async = true; // 可选：异步加载，不阻塞页面
					// script.defer = true;

					script.onload = () => {
						console.log(`${src} 加载完成`);
						resolve(script);
					};

					script.onerror = () => {
						reject(new Error(`加载 ${src} 失败`));
					};

					document.head.appendChild(script);
				});
			}

			// 使用示例：
			loadScript('./js/b.js')
			.then(() => {
				// ✅ b.js 已加载完成，可以安全调用 b.js 中定义的函数或变量
				console.log('开始执行依赖 b.js 的代码');

				// 比如调用 b.js 中的函数：
				// someFunctionFromB();
			})
			.catch(err => {
				console.error(err);
			});

注意事项：

  1. b.js 中的代码必须是“可被后续访问”的
    • 例如：在 b.js 中定义了全局函数或变量：
      • 
					// b.js
      
					function someFunctionFromB() {
      
						console.log('Function from b.js');
      
					}
    • 或者使用模块导出（但动态 script 标签不支持 ES Module 语法 export，除非用 type="module"，见下文）。
  2. 如果 b.js 是 ES Module，需要加 type="module"
    3. 
				script.type = 'module';

    注意：模块脚本默认是 defer 的，且不能使用 document.write。

  3. 不要用 document.write 或 innerHTML += '<script>'
    • 这种方式不可靠，且可能阻塞或破坏页面结构。

高级封装：加载多个脚本并顺序执行


			function loadScript(src) {
				return new Promise((resolve, reject) => {
					const script = document.createElement('script');
					script.src = src;
					script.onload = () => resolve(script);
					script.onerror = () => reject(new Error(`Script load error: ${src}`));
					document.head.appendChild(script);
				});
			}

			// 顺序加载多个脚本
			loadScript('b1.js')
			.then(() => loadScript('b2.js'))
			.then(() => loadScript('b3.js'))
			.then(() => {
				console.log('所有脚本加载完成');
				// 执行你的逻辑
			})
			.catch(console.error);

高级封装：加载多个脚本并顺序执行 - async/await


			function loadScript(src) {
				return new Promise((resolve, reject) => {
					const script = document.createElement('script');
					script.src = src;
					script.onload = () => resolve(script);
					script.onerror = () => reject(new Error(`Script load error: ${src}`));
					document.head.appendChild(script);
				});
			}

			async function init() {
				try {
					await loadScript('b1.js');
					await loadScript('b2.js');
					await loadScript('b3.js');
					console.log('全部加载完成');
					// 执行后续逻辑
				} catch (err) {
					console.error(err);
				}
			}

			init();
